from turtle import*
import turtle
t = turtle.Turtle()

def draw_circle(color,radius,x,y):

    t.penup()
    t.fillcolor(color)
    t.goto(x,y)
    t.pendown()
    t.begin_fill()
    t.circle(radius)
    t.end_fill()
