Mandala me rrathë koncentrikë në 5 ngjyra.

🐍 Kodi Python origjinalProject (EX) 11.py
import turtle

wn = turtle.Screen()
wn.bgcolor('black')
s = turtle.Turtle()
s.speed('fastest')
s.color('white')
rotate = int(180)


def Circles(t, size):
    for i in range(10):
        t.circle(size)
        size = size - 4


def vishal(t, size, repeat):
    for i in range(repeat):
        Circles(t, size)
        t.right(360 / repeat)


vishal(s, 200, 10)
